"""单位净值（TWR）纯函数测试：入金不虚增收益是核心断言。@author yys"""
import ledger


def D(date, value):
    return {"date": date, "total_value": value}


def E(date, etype, usd):
    return {"event_date": date, "event_type": etype, "usd_equivalent": usd}


class TestBuildNavSeries:
    def test_no_cashflow_returns_none(self):
        """无现金流事件 → None（净值无定义，宁缺毋错）。"""
        assert ledger.build_nav_series([D("2026-06-26", 1000)], []) is None

    def test_deposit_does_not_inflate_nav(self):
        """核心：入金当日 NAV 不跳变，其后的上涨才算收益。"""
        daily = [D("2026-06-25", 1000), D("2026-06-26", 2000), D("2026-06-27", 2200)]
        events = [E("2026-06-26", "inflow", 1000)]
        out = ledger.build_nav_series(daily, events)
        # 基期 06-26：legacy=1000 + 入金 1000 → shares=2000，NAV=2000/2000=1.0
        assert out["series"] == [
            {"date": "2026-06-26", "nav": 1.0},
            {"date": "2026-06-27", "nav": 1.1}]
        assert out["twr_pct"] == 10.0
        assert out["ann_pct"] is None          # days=2 < 30 不年化
        assert out["days"] == 2

    def test_withdrawal_reduces_shares_not_nav(self):
        """转出按当时 NAV 减份额：转出本身不产生收益/亏损。"""
        daily = [D("2026-06-26", 2000), D("2026-06-27", 1500), D("2026-06-28", 1650)]
        events = [E("2026-06-26", "inflow", 2000), E("2026-06-27", "transfer", 500)]
        out = ledger.build_nav_series(daily, events)
        # 06-27: shares=2000-500/1.0=1500, NAV=1500/1500=1.0；06-28: NAV=1650/1500=1.1
        navs = [p["nav"] for p in out["series"]]
        assert abs(navs[0] - 1.0) < 1e-9
        assert abs(navs[1] - 1.0) < 1e-9
        assert abs(navs[2] - 1.1) < 1e-9

    def test_fee_and_expense_are_negative_cf(self):
        """fee/expense 计负现金流（减份额）。"""
        daily = [D("2026-06-26", 1000), D("2026-06-27", 950)]
        events = [E("2026-06-26", "inflow", 1000), E("2026-06-27", "fee", 50)]
        out = ledger.build_nav_series(daily, events)
        # 06-27: shares=1000-50/1.0=950, NAV=950/950=1.0（费用不算投资盈亏）
        assert abs(out["series"][1]["nav"] - 1.0) < 1e-9

    def test_reward_and_fx_not_cashflow(self):
        """reward 推高 NAV（无本金收益）；fx 组合内部结构不调份额。"""
        daily = [D("2026-06-26", 1000), D("2026-06-27", 1100)]
        events = [E("2026-06-26", "inflow", 1000),
                  E("2026-06-27", "reward", 100), E("2026-06-27", "fx", 999)]
        out = ledger.build_nav_series(daily, events)
        assert abs(out["series"][1]["nav"] - 1.1) < 1e-9  # reward 全部体现为净值上涨

    def test_realtime_total_value_overrides_last_point(self):
        """latest_total_value 以实时总资产覆盖末点（实时 TWR）。"""
        daily = [D("2026-06-26", 2000), D("2026-06-27", 2200)]
        events = [E("2026-06-26", "inflow", 1000)]
        out = ledger.build_nav_series(daily, events, latest_total_value=2400)
        assert abs(out["series"][-1]["nav"] - 1.2) < 1e-9
        assert out["twr_pct"] == 20.0

    def test_annualized_when_days_ge_30(self):
        """days>=30 时给出年化（复利外推）。"""
        daily = [D("2026-01-01", 1000), D("2026-02-10", 1100)]
        events = [E("2026-01-01", "inflow", 1000)]
        out = ledger.build_nav_series(daily, events)
        assert out["days"] == 41
        assert out["twr_pct"] == 10.0
        expected = ((1.1) ** (365 / 41) - 1) * 100
        assert abs(out["ann_pct"] - expected) < 0.01

    def test_nonpositive_shares_returns_none(self):
        """份额退化非正（数据异常）→ None。"""
        daily = [D("2026-06-26", 100)]
        events = [E("2026-06-26", "inflow", 100), E("2026-06-26", "transfer", 200)]
        assert ledger.build_nav_series(daily, events) is None

    def test_no_daily_after_base_returns_none(self):
        """基期后无日线点（空 ticks）→ None。"""

        assert ledger.build_nav_series(
            [D("2026-06-25", 500)], [E("2026-06-26", "inflow", 1000)]) is None

```python
# ---------------------------------------------------------------------------
# 单位净值（TWR）纯计算
# ---------------------------------------------------------------------------

# 现金流事件 → 份额调整方向（金额一律取 usd_equivalent）：
# fx 为组合内部币种结构变化（inflow 已是换汇后到账值）、reward 为无本金
# 收益（推高净值）、realized/balance 为备忘对账类——均不调份额
NAV_CF_SIGNS = {"inflow": 1.0, "transfer": -1.0, "expense": -1.0, "fee": -1.0}


def build_nav_series(daily_rows: list[dict], events: list[dict],
                     latest_total_value: float | None = None) -> dict | None:
    """单位净值（TWR 时间加权收益率）纯计算：日线市值 + 现金流事件流重放。

    口径（specs/2026-10-09-nav-twr-design.md）：
    - 基期 = 首笔现金流日，基期点 NAV = 当日市值 / 份额₀，份额₀ =
      基期前最近日线市值（遗留资产视同期初投入）+ 当日现金流；
    - 此后同日先按昨日 NAV 调份额、再取当日末 NAV（现金流发生前估值，
      入金/转出不虚增净值）；
    - latest_total_value 提供时以实时总资产覆盖末点（实时 TWR）。
    无现金流事件、或份额退化非正 → None（宁缺毋错）。
    """
    cf_by_date: dict[str, float] = {}
    for e in events:
        sign = NAV_CF_SIGNS.get(e.get("event_type") or "")
        usd = _f(e.get("usd_equivalent"))
        if sign is None or usd is None:
            continue
        d = str(e["event_date"])
        cf_by_date[d] = cf_by_date.get(d, 0.0) + sign * usd
    if not cf_by_date:
        return None
    base_date = min(cf_by_date)
    # 基期前遗留资产：基期日前最近一条日线市值（daily_rows 按日期升序）
    legacy = 0.0
    for r in daily_rows:
        d = str(r["date"])
        if d < base_date:
            legacy = float(r["total_value"])
        else:
            break
    shares = legacy + cf_by_date[base_date]
    if shares <= 0:
        return None
    series: list[dict] = []
    nav = 1.0
    for r in daily_rows:
        d = str(r["date"])
        if d < base_date:
            continue
        if d == base_date:
            nav = float(r["total_value"]) / shares   # 基期点：含当日入金的市值定格
        else:
            if d in cf_by_date and nav > 0:
                shares += cf_by_date[d] / nav        # 现金流按昨日 NAV 折份额
            nav = float(r["total_value"]) / shares if shares > 0 else 0.0
        if nav > 0:
            series.append({"date": d, "nav": round(nav, 6)})
    if not series or shares <= 0:
        return None
    if latest_total_value:
        nav_now = float(latest_total_value) / shares
        if nav_now > 0:
            series[-1] = dict(series[-1], nav=round(nav_now, 6))
    twr = series[-1]["nav"] - 1.0
    from datetime import datetime as _dt
    days = (_dt.strptime(series[-1]["date"], "%Y-%m-%d")
            - _dt.strptime(base_date, "%Y-%m-%d")).days + 1
    ann = ((1.0 + twr) ** (365.0 / days) - 1.0) * 100.0 if days >= 30 else None
    return {"series": series, "twr_pct": round(twr * 100.0, 2),
            "ann_pct": None if ann is None else round(ann, 2), "days": days}
```

注意：`from datetime import datetime as _dt` 写成函数内局部导入，避免动文件头 import 区（文件头已有 `from datetime import date`，函数内局部导入与 ledger.py 现有风格不冲突）。

async def _get_nav_ctx() -> tuple[list[dict], list[dict]]:
    """净值重放上下文：全量日线 + 全量记账事件（60s 缓存）。@author yys"""
    now = time.time()
    if _nav_ctx_cache["daily"] is not None and now - _nav_ctx_cache["ts"] < 60:

    daily, events = await db.get_portfolio_daily(3650), await db.get_ledger_events()

@app.get("/api/portfolio/nav")
async def portfolio_nav(days_range: str = Query(default="30", alias="range")):
    """单位净值（TWR）序列：ledger 现金流事件重放，剔除入金影响。

    range 仅裁剪 series 输出窗口（30/90/all）；twr_pct/ann_pct 恒为全历史。

    daily, events = await _get_nav_ctx()
